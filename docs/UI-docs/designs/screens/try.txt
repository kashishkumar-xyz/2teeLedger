// Custom Colors based on tailwind.config and inline styles
val Primary = Color(0xFF39E079)
val ForensicRed = Color(0xFF8B0000)
val ForensicDark = Color(0xFF0A0A0A)
val ForensicLight = Color(0xFF1A1A1A)
val CustomGreen = Color(0xFF39E079) // Same as Primary

// Grayscale colors (approximations based on common Tailwind values and context)
val Gray200 = Color(0xFFE0E0E0)
val Gray400 = Color(0xFF9CA3AF)
val Gray500 = Color(0xFF6B7280)
val Gray700 = Color(0xFF374151)

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MaterialTheme {
                StitchDesignScreen()
            }
        }
    }
}

enum class OperationType {
    SUBTRACT,
    TRANSFER,
    ADD
}

enum class FooterTab {
    RECENTS,
    KEYPAD,
    ACCOUNTS
}

@Composable
fun StitchDesignScreen() {
    var isNegative by remember { mutableStateOf(false) }
    var selectedOperation by remember { mutableStateOf(OperationType.TRANSFER) }
    var selectedTab by remember { mutableStateOf(FooterTab.KEYPAD) }

    val currentPrimaryColor = if (isNegative) ForensicRed else Primary

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(ForensicDark)
            .statusBarsPadding()
            .navigationBarsPadding(),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        // Header
        Box(modifier = Modifier.fillMaxWidth().wrapContentHeight()) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Spacer(modifier = Modifier.width(48.dp))
                Row(
                    modifier = Modifier.width(48.dp),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.End
                ) {
                    Button(
                        onClick = { /* settings action */ },
                        modifier = Modifier.size(48.dp),
                        shape = CircleShape,
                        colors = ButtonDefaults.buttonColors(containerColor = Color.Transparent),
                        contentPadding = PaddingValues(0.dp)
                    ) {
                        Icon(
                            imageVector = Icons.Filled.Settings,
                            contentDescription = "Settings",
                            tint = Gray400,
                            modifier = Modifier.size(24.dp)
                        )
                    }
                }
            }
        }

        // Main Content
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .weight(1f)
                .padding(horizontal = 16.dp)
                .padding(bottom = 8.dp),
            verticalArrangement = Arrangement.SpaceEvenly
        ) {
            // Amount Display & Note
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.spacedBy(16.dp)
            ) {
                // Amount
                Box(modifier = Modifier.fillMaxWidth()) {
                    Text(
                        text = buildAnnotatedString {
                            withStyle(style = SpanStyle(color = currentPrimaryColor)) {
                                append("$")
                            }
                            withStyle(style = SpanStyle(color = currentPrimaryColor.copy(alpha = 0.9f))) {
                                append("0")
                            }
                        },
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(vertical = 8.dp),
                        textAlign = TextAlign.Center,
                        fontSize = 72.sp,
                        fontWeight = FontWeight.Bold
                    )
                }

                // Add note section
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 8.dp)
                        .padding(top = 16.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .background(ForensicLight.copy(alpha = 0.5f), RoundedCornerShape(8.dp))
                            .padding(horizontal = 12.dp, vertical = 8.dp),
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.SpaceBetween
                    ) {
                        Text("Add note", color = Gray400, fontSize = 14.sp)
                        Text("Add details...", color = Color.White.copy(alpha = 0.5f), fontSize = 14.sp, fontWeight = FontWeight.Medium)
                    }
                }
            }

            // Keypad Grid
            val keypadGridColor = Color.White.copy(alpha = 0.05f)
            val keypadGridSize = 20.dp
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp)
                    .background(Color.Transparent, RoundedCornerShape(12.dp))
                    .drawBehind {
                        val cellSizePx = keypadGridSize.toPx()
                        for (x in cellSizePx.rangeTo(size.width).step(cellSizePx.toLong())) {
                            drawLine(keypadGridColor, Offset(x, 0f), Offset(x, size.height), strokeWidth = 1.dp.toPx())
                        }
                        for (y in cellSizePx.rangeTo(size.height).step(cellSizePx.toLong())) {
                            drawLine(keypadGridColor, Offset(0f, y), Offset(size.width, y), strokeWidth = 1.dp.toPx())
                        }
                    },
                verticalArrangement = Arrangement.spacedBy(16.dp)
            ) {
                // Number buttons grid
                Column(verticalArrangement = Arrangement.spacedBy(16.dp)) {
                    val buttons = listOf(
                        listOf("1", "2", "3"),
                        listOf("4", "5", "6"),
                        listOf("7", "8", "9"),
                        listOf("C", "0", "backspace")
                    )
                    buttons.forEach { rowButtons ->
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.spacedBy(16.dp)
                        ) {
                            rowButtons.forEach { buttonText ->
                                val fontSize = when (buttonText) {
                                    "C" -> 20.sp
                                    "backspace" -> 30.sp // Icon size
                                    else -> 30.sp
                                }
                                Button(
                                    onClick = { /* keypad button action */ },
                                    modifier = Modifier
                                        .weight(1f)
                                        .aspectRatio(1f)
                                        .heightIn(min = 80.dp),
                                    shape = CircleShape,
                                    colors = ButtonDefaults.buttonColors(containerColor = ForensicLight),
                                    border = BorderStroke(1.dp, CustomGreen.copy(alpha = 0.2f)),
                                    elevation = ButtonDefaults.buttonElevation(defaultElevation = 4.dp),
                                    contentPadding = PaddingValues(0.dp)
                                ) {
                                    if (buttonText == "backspace") {
                                        Icon(
                                            imageVector = Icons.Filled.Backspace,
                                            contentDescription = "Backspace",
                                            tint = Color.White,
                                            modifier = Modifier.size(fontSize)
                                        )
                                    } else {
                                        Text(buttonText, fontSize = fontSize, fontWeight = FontWeight.Bold, color = Color.White)
                                    }
                                }
                            }
                        }
                    }
                }

                // Operation buttons
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(top = 16.dp),
                    horizontalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    OperationButton(
                        text = "-",
                        isActive = selectedOperation == OperationType.SUBTRACT,
                        onClick = {
                            selectedOperation = OperationType.SUBTRACT
                            isNegative = true
                        }
                    )
                    OperationButton(
                        text = ">",
                        isActive = selectedOperation == OperationType.TRANSFER,
                        onClick = {
                            selectedOperation = OperationType.TRANSFER
                            isNegative = false
                        }
                    )
                    OperationButton(
                        text = "+",
                        isActive = selectedOperation == OperationType.ADD,
                        onClick = {
                            selectedOperation = OperationType.ADD
                            isNegative = false
                        }
                    )
                }
            }
        }

        // Footer
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .wrapContentHeight()
                .background(ForensicDark)
        ) {
            Spacer(modifier = Modifier.fillMaxWidth().height(1.dp).background(Gray700))
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, top = 8.dp, bottom = 20.dp),
                horizontalArrangement = Arrangement.SpaceAround
            ) {
                FooterNavigationItem(
                    icon = Icons.Filled.History,
                    label = "Recents",
                    isSelected = selectedTab == FooterTab.RECENTS,
                    onClick = { selectedTab = FooterTab.RECENTS }
                )
                FooterNavigationItem(
                    icon = Icons.Filled.Dialpad,
                    label = "Keypad",
                    isSelected = selectedTab == FooterTab.KEYPAD,
                    onClick = { selectedTab = FooterTab.KEYPAD }
                )
                FooterNavigationItem(
                    icon = Icons.Filled.Person,
                    label = "Accounts",
                    isSelected = selectedTab == FooterTab.ACCOUNTS,
                    onClick = { selectedTab = FooterTab.ACCOUNTS }
                )
            }
        }
    }
}

@Composable
fun OperationButton(text: String, isActive: Boolean, onClick: () -> Unit) {
    Button(
        onClick = onClick,
        modifier = Modifier
            .weight(1f)
            .height(60.dp),
        shape = RoundedCornerShape(12.dp),
        colors = ButtonDefaults.buttonColors(
            containerColor = if (isActive) CustomGreen else ForensicLight
        ),
        border = if (isActive) BorderStroke(2.dp, CustomGreen) else null,
        contentPadding = PaddingValues(0.dp)
    ) {
        Text(text = text, fontSize = 18.sp, fontWeight = FontWeight.Bold, color = Color.White)
    }
}

@Composable
fun FooterNavigationItem(icon: ImageVector, label: String, isSelected: Boolean, onClick: () -> Unit) {
    val color = if (isSelected) CustomGreen else Gray500

    Column(
        modifier = Modifier
            .weight(1f)
            .fillMaxHeight()
            .clickable { onClick() }
            .padding(vertical = 4.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(4.dp)
    ) {
        Icon(
            imageVector = icon,
            contentDescription = label,
            tint = color,
            modifier = Modifier.size(24.dp)
        )
        Text(
            text = label,
            fontSize = 12.sp,
            fontWeight = FontWeight.Medium,
            color = color
        )
    }
}

@Preview(showBackground = true)

fun DefaultPreview() {
    MaterialTheme {
        StitchDesignScreen()
    }
